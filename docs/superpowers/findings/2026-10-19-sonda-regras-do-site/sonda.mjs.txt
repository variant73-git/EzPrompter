async function coletarDoExtrator() {
  const src = await readFile(new URL('./scripts/inventario-conteudo.mjs', import.meta.url), 'utf8');
  const i = src.indexOf('function coletar(origem) {'); const j = src.indexOf('\nexport async function inventariar');
  let corpo = src.slice(i, j);
  corpo = corpo.replace(/chave: chave\('([a-z])'\)/g, "chave: window.__marca(el, chave('$1'))");
  corpo = corpo.replace(/chave: `l-\$\{String\(out\.lotties\.length \+ 1\)\.padStart\(4, '0'\)\}`/g, "chave: window.__marca(el, `l-${String(out.lotties.length + 1).padStart(4, '0')}`)");
  return corpo;
}

// ---------- NO NAVEGADOR: regras do site e quem cada uma pega ----------
function coletarRegras() {
  const todos = [document.documentElement, ...document.querySelectorAll('*')];
  todos.forEach((el, k) => el.setAttribute('data-u-k', String(k)));
  const est = { folhas: 0, inacessiveis: [], fontFace: 0, keyframes: 0, outrasAt: 0, aninhadas: 0, estadoAncestral: 0, invalidos: 0, seletores: 0, prelude: [] };
  const entradas = [];
  const dividir = (s) => { const out = []; let d = 0, q = null, cur = ''; for (const ch of s) { if (q) { cur += ch; if (ch === q) q = null; continue; } if (ch === '"' || ch === "'") { q = ch; cur += ch; continue; } if ('([' .includes(ch)) d += 1; if (')]'.includes(ch)) d -= 1; if (ch === ',' && d === 0) { out.push(cur.trim()); cur = ''; continue; } cur += ch; } if (cur.trim()) out.push(cur.trim()); return out; };
  // composto final (depois do último combinador fora de parênteses/colchetes)
  const ultimoComposto = (s) => { let d = 0, q = null, i0 = 0; for (let i = 0; i < s.length; i += 1) { const ch = s[i]; if (q) { if (ch === q) q = null; continue; } if (ch === '"' || ch === "'") { q = ch; continue; } if ('(['.includes(ch)) d += 1; else if (')]'.includes(ch)) d -= 1; else if (d === 0 && /[\s>+~]/.test(ch)) i0 = i + 1; } return i0; };
  const PSEUDO_EL = /(::?(?:before|after|first-line|first-letter)|::(?:placeholder|selection|marker|backdrop|file-selector-button|-webkit-[\w-]+|-moz-[\w-]+))$/i;
  const ESTADO = /:(?:hover|focus-visible|focus-within|focus|active|visited|target)(?![\w-])/gi;
  // especificidade (a,b,c) de um seletor complexo — aproximada o bastante para a sonda
  const espec = (s) => {
    let a = 0, b = 0, c = 0; let t = s;
    t = t.replace(/:where\((?:[^()]|\([^()]*\))*\)/gi, '');
    t = t.replace(/:(is|not|has|matches)\(((?:[^()]|\([^()]*\))*)\)/gi, (m, f, arg) => { let mx = [0, 0, 0]; for (const x of dividir(arg)) { const e = espec(x); if (e[0] > mx[0] || (e[0] === mx[0] && (e[1] > mx[1] || (e[1] === mx[1] && e[2] > mx[2])))) mx = e; } a += mx[0]; b += mx[1]; c += mx[2]; return ' '; });
    t = t.replace(/\[[^\]]*\]/g, () => { b += 1; return ' '; });
    t = t.replace(/#[\w-]+/g, () => { a += 1; return ' '; });
    t = t.replace(/::[\w-]+(\([^)]*\))?/g, () => { c += 1; return ' '; });
    t = t.replace(/:(before|after|first-line|first-letter)(?![\w-])/gi, () => { c += 1; return ' '; });
    t = t.replace(/:[\w-]+(\([^)]*\))?/g, () => { b += 1; return ' '; });
    t = t.replace(/\.[\w-]+/g, () => { b += 1; return ' '; });
    t.replace(/(^|[\s>+~(])([a-zA-Z][\w-]*)/g, () => { c += 1; return ''; });
    return [a, b, c];
  };
  const andar = (regras, cadeia, base) => {
    for (const r of Array.from(regras)) {
      const tipo = r.constructor.name;
      if (tipo === 'CSSStyleRule') {
        if (r.cssRules && r.cssRules.length) est.aninhadas += 1;
        for (const sel of dividir(r.selectorText)) {
          est.seletores += 1;
          const pe = (sel.match(PSEUDO_EL) || [''])[0]; let semPe = pe ? sel.slice(0, -pe.length) : sel;
          const i0 = ultimoComposto(semPe);
          if (ESTADO.test(semPe.slice(0, i0))) { est.estadoAncestral += 1; ESTADO.lastIndex = 0; continue; }
          ESTADO.lastIndex = 0;
          const estados = (semPe.slice(i0).match(ESTADO) || []).join('');
          let alvo = semPe.slice(0, i0) + semPe.slice(i0).replace(ESTADO, ''); ESTADO.lastIndex = 0;
          if (!alvo.trim() || /[\s>+~]$/.test(alvo)) alvo = alvo + '*';
          let ks;
          try { ks = Array.from(document.querySelectorAll(alvo)).map((e) => Number(e.getAttribute('data-u-k'))).filter((n) => !Number.isNaN(n)); } catch (e) { est.invalidos += 1; continue; }
          entradas.push({ cadeia, sel, pe, estados, ks, espec: espec(sel), decl: r.style.cssText, base });
        }
      } else if (tipo === 'CSSImportRule') {
        if (r.styleSheet) { const m = r.media && r.media.mediaText; andar(lerRegras(r.styleSheet), m ? [...cadeia, `@media ${m}`] : cadeia, r.styleSheet.href || base); }
      } else if (tipo === 'CSSFontFaceRule') est.fontFace += 1;
      else if (tipo === 'CSSKeyframesRule') est.keyframes += 1;
      else if (tipo === 'CSSLayerStatementRule' || tipo === 'CSSPropertyRule') { if (!cadeia.length) est.prelude.push(r.cssText); }
      else if (r.cssRules) { const h = r.cssText.slice(0, r.cssText.indexOf('{')).trim(); andar(r.cssRules, [...cadeia, h], base); }
      else est.outrasAt += 1;
    }
  };
  const lerRegras = (folha) => { try { return folha.cssRules; } catch (e) { est.inacessiveis.push(folha.href || '(inline)'); return []; } };
  for (const f of [...document.styleSheets, ...(document.adoptedStyleSheets || [])]) { est.folhas += 1; andar(lerRegras(f), [], f.href || location.href); }
  const inline = {}; for (const el of todos) { const v = (el.getAttribute('style') || '').trim(); if (v) inline[el.getAttribute('data-u-k')] = v; }
  est.elementosComStyle = Object.keys(inline).length;
  return { entradas, est, total: todos.length, inline };
}

// ---------- NO NAVEGADOR: caixa de layout por id (sem transformações: cadeia de offsets) ----------
function caixas(atributo) {
  const o = {};
  for (const el of document.querySelectorAll(`[${atributo}]`)) {
    if (!(el instanceof HTMLElement)) continue;
    const cs = getComputedStyle(el);
    if (cs.display === 'none' || cs.visibility === 'hidden' || !el.offsetWidth || !el.offsetHeight) continue;
    let x = 0, y = 0; for (let e = el; e; e = e.offsetParent) { x += e.offsetLeft; y += e.offsetTop; }
    o[el.getAttribute(atributo)] = [x, y, el.offsetWidth, el.offsetHeight];
  }
  return { caixas: o, largura: document.documentElement.scrollWidth, janela: innerWidth, altura: document.documentElement.scrollHeight };
}
// elementos com conteúdo próprio VISÍVEIS nesta tela que NÃO estavam visíveis no desktop e não têm id na cópia:
// existem só nesta tela (menu de celular, versão mobile de uma seção) — a cópia de um documento só não os tem
function soNestaTela(marcar) {
  let n = 0;
  for (const el of document.body.querySelectorAll('*')) {
    if (!(el instanceof HTMLElement) || el.closest('svg')) continue;
    if (!marcar && (el.hasAttribute('data-u-id') || el.hasAttribute('data-u-vis'))) continue;
    if (/^(SCRIPT|STYLE|LINK|META|NOSCRIPT|TEMPLATE|BR)$/.test(el.tagName)) continue;
    const cs = getComputedStyle(el); if (cs.display === 'none' || cs.visibility === 'hidden' || !el.offsetWidth || !el.offsetHeight) continue;
    if (marcar) { el.setAttribute('data-u-vis', ''); continue; }
    // só conta quem tem CONTEÚDO próprio (texto direto ou mídia) — embrulho desfeito não é perda
    const texto = Array.from(el.childNodes).some((c) => c.nodeType === 3 && c.textContent.trim());
    if (texto || /^(IMG|VIDEO|CANVAS|PICTURE)$/.test(el.tagName)) n += 1;
  }
  return n;
}

// ---------- medir as duas cópias contra o original ----------
const comparar = (nat, cop) => {
  const ids = Object.keys(nat.caixas); let presente = 0, tamanho = 0, x = 0, y = 0;
  for (const id of ids) {
    const n = nat.caixas[id]; const c = cop.caixas[id]; if (!c) continue; presente += 1;
    if (Math.abs(n[2] - c[2]) <= Math.max(4, 0.02 * n[2]) && Math.abs(n[3] - c[3]) <= Math.max(4, 0.02 * n[3])) tamanho += 1;
    if (Math.abs(n[0] - c[0]) <= 4) x += 1;
    if (Math.abs(n[1] - c[1]) <= 8) y += 1;
  }
  const pct = (v) => `${Math.round((100 * v) / Math.max(1, ids.length))}%`;
  return { elementos: ids.length, presentes: pct(presente), tamanhoIgual: pct(tamanho), xIgual: pct(x), yIgual: pct(y), transborda: cop.largura > cop.janela + 1, alturaCopia: cop.altura, alturaOriginal: nat.altura };
};
